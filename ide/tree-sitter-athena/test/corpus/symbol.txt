========================================
Mixed Symbol and Element with composition
========================================

package athena.iec

symbol iec_switch_contact {
  identity "iec.switch_contact"
  version "1.0.0"
}

element iec_switch_module {
  identity "iec.switch_module"
  version "1.0.0"
  bounds (0, 0, 180, 80)

  child primary {
    symbol "iec.switch_contact"
    translate (0, 0)
    rotate 0
    scale (1, 1)
    zOrder 0
  }

  child auxiliary {
    symbol "iec.switch_contact"
    translate (100, 0)
    rotate 0
    scale (1, 1)
    zOrder 1
  }

  export anchor primaryLine from primary.line
  export anchor primaryLoad from primary.load
  export anchor auxiliaryLine from auxiliary.line
  export anchor auxiliaryLoad from auxiliary.load
}

package athena.iec

symbol coil {
  identity "iec.coil.symbol"
  version "1.0.0"
  graphic {
    bounds (0, 0, 80, 80)
    line lead from (40, 0) to (40, 14) style conductor
    polyline marker points ((10, 10), (20, 20), (30, 10)) style reference
    arc winding center (40, 40) radius 12 from 180 sweep 180 style symbol
    circle terminalMark center (40, 0) radius 2 style terminal
    rectangle body at (20, 14) size (40, 52) style symbol
    label deviceTag at (0, -14) size (80, 12) role device-tag style device-label
  }
}

element coil_element {
  identity "iec.coil.element"
  version "1.0.0"
  bounds (0, -14, 80, 94)
  child glyph { symbol "iec.coil.symbol" translate (0, 0) rotate 0 scale (1, 1) zOrder 0 }
  export label deviceTag from glyph.deviceTag
}

profile IECProfile {
  projection schematic
  standard IEC
  style industrial-iec-v1
  fallback fail-closed
}

binding CoilBinding {
  profile IECProfile
  priority 200
  select function where { type Contactor role coil }
  use element "iec.coil.element" version "1.0.0"
  variant "standard"
}
